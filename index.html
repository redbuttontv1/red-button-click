<script>
const SUPABASE_URL = "PASTE_YOUR_URL_HERE";
const SUPABASE_KEY = "PASTE_YOUR_PUBLIC_KEY_HERE";
let deviceID = localStorage.getItem("deviceID");
if (!deviceID) {
deviceID = crypto.randomUUID();
localStorage.setItem("deviceID", deviceID);
}
const button = document.getElementById("button");
const total = document.getElementById("total");
const mine = document.getElementById("mine");
// Load global total
async function loadTotal() {
const response = await fetch(
SUPABASE_URL + "/rest/v1/button_stats?id=eq.1",
{
headers:{
"apikey": SUPABASE_KEY
}
}
);
const data = await response.json();
if(data[0]){
total.textContent = data[0].total_clicks;
}
}
// Load this device's clicks
async function loadMyClicks(){
const response = await fetch(
SUPABASE_URL + "/rest/v1/device_clicks?device_id=eq." + deviceID,
{
headers:{
"apikey": SUPABASE_KEY
}
}
);
const data = await response.json();
if(data.length > 0){
mine.textContent = data[0].clicks;
} else {
await fetch(
SUPABASE_URL + "/rest/v1/device_clicks",
{
method:"POST",
headers:{
"apikey":SUPABASE_KEY,
"Content-Type":"application/json"
},
body:JSON.stringify({
device_id:deviceID,
clicks:0
})
}
);
mine.textContent = 0;
}
}
// Button press
button.onclick = async ()=>{
button.classList.toggle("glow");
let current = Number(total.textContent);
current++;
// Update global count
await fetch(
SUPABASE_URL + "/rest/v1/button_stats?id=eq.1",
{
method:"PATCH",
headers:{
"apikey":SUPABASE_KEY,
"Content-Type":"application/json"
},
body:JSON.stringify({
total_clicks:current
})
}
);
// Update device count
let myCurrent = Number(mine.textContent);
myCurrent++;
await fetch(
SUPABASE_URL + "/rest/v1/device_clicks?device_id=eq." + deviceID,
{
method:"PATCH",
headers:{
"apikey":SUPABASE_KEY,
"Content-Type":"application/json"
},
body:JSON.stringify({
clicks:myCurrent
})
}
);
total.textContent=current;
mine.textContent=myCurrent;
};
// Initial load
loadTotal();
loadMyClicks();
// Refresh total automatically
setInterval(loadTotal,3000);
</script>
